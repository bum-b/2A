        // ===== NAVIGATION ENGINE =====
        function showPage(pageId) {
            // Hide all pages
            document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
            // Show selected page
            document.getElementById(pageId).classList.add('active');
            // Reset menu view if navigating to menu page
            if(pageId === 'menu-view') backToFolders();
            // Scroll to top
            window.scrollTo(0,0);
        }

        // ===== CUISINE FOLDER LOGIC =====
        function turnPageAndOpen(folderElement, cuisineName) {
            folderElement.classList.add('page-turn');
            setTimeout(() => {
                folderElement.classList.remove('page-turn');
                openCuisine(cuisineName);
            }, 200);
        }

        function openCuisine(cuisineName) {
            document.getElementById('cuisine-folder-view').style.display = 'none';
            document.getElementById('cuisine-items-view').style.display = 'block';
            document.getElementById('current-cuisine-title').innerText = cuisineName + " Cuisine";

            const target = document.getElementById('menu-content-target');
            target.innerHTML = '';

            const categories = masterMenu[cuisineName];
            let rowIndex = 0;
            for(let catName in categories) {
                let html = `<h3 class="menu-cat">${catName}</h3>`;
                for(let itemName in categories[catName]) {
                    html += `<div class="menu-card" style="animation-delay: ${rowIndex * 0.05}s"><h4>${itemName}</h4><div class="price">₹${categories[catName][itemName]}</div><p>From our ${catName} collection, crafted for flavor.</p></div>`;
                    rowIndex++;
                }
                target.innerHTML += html;
            }
            window.scrollTo(0,0);
        }

        function backToFolders() {
            document.getElementById('cuisine-items-view').style.display = 'none';
            document.getElementById('cuisine-folder-view').style.display = 'block';

            let trailPositions = Array(trails.length).fill({ x: 0, y: 0 });
            let retractionTimeout;

            // ===== CURSOR TRAIL LOGIC =====
            document.addEventListener('mousemove', (e) => {
                mouseX = e.clientX;
                mouseY = e.clientY;

                lens.style.left = `${e.clientX - 100}px`;
                lens.style.top = `${e.clientY - 100}px`;
                cursor.style.transform = `translate3d(${e.clientX}px, ${e.clientY}px, 0)`;

                // Clear existing retraction
                clearTimeout(retractionTimeout);
                trails.forEach(trail => trail.style.opacity = '1');

                // Update trail positions with delay
                trailPositions.unshift({ x: e.clientX, y: e.clientY });
                trailPositions = trailPositions.slice(0, trails.length);

                trails.forEach((trail, index) => {
                    const pos = trailPositions[index + 1] || { x: e.clientX, y: e.clientY };
                    setTimeout(() => {
                        trail.style.transform = `translate3d(${pos.x}px, ${pos.y}px, 0)`;
                    }, index * 50);
                });

                // Start retraction after 500ms of no movement
                retractionTimeout = setTimeout(() => {
                    trails.forEach((trail, index) => {
                        setTimeout(() => {
                            trail.style.opacity = '0';
                        }, index * 100);
                    });
                }, 500);
            });

            // ===== HOVER EFFECTS =====
            // Apply blur on hover to folders
            document.querySelectorAll('.folder').forEach((folder) => {
                folder.classList.add('blur-on-hover');
                folder.addEventListener('mouseenter', () => cursor.classList.add('cursor--active'));
                folder.addEventListener('mouseleave', () => cursor.classList.remove('cursor--active'));
            });

            // Parallax scroll effect for home view
            window.addEventListener('scroll', () => {
                const scrolled = window.pageYOffset;
                const homeView = document.getElementById('home-view');
                if (homeView.classList.contains('active')) {
                    homeView.style.backgroundPositionY = `${-scrolled * 0.5}px`;
                }
            });

            document.querySelectorAll('a, button, .folder, .back-btn, .submit-btn').forEach(el => {
                el.addEventListener('mouseenter', () => cursor.classList.add('cursor--hover'));
                el.addEventListener('mouseleave', () => cursor.classList.remove('cursor--hover'));
            });

            // Flicker effect on logo
            logo.classList.add('flicker');

            // ===== SCROLL ANIMATIONS =====
            // Reveal review cards on scroll
            const observer = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        entry.target.style.animationPlayState = 'running';
                    }
                });
            }, { threshold: 0.1, rootMargin: '0px 0px -40px 0px' });

            document.querySelectorAll('.review-card').forEach(card => {
                card.style.animation = 'slideInUp 0.8s ease-out paused';
                observer.observe(card);
            });
        });
    </script>
</body>
